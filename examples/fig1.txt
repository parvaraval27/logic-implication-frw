// Paper Fig. 1 - Example circuit
// Implications: e(=1) => G8(=0), b(=1) => G8(=0), h(=0) => G8(=0)
// Gates: G1-G10, Inputs: a-h, Outputs: O1, O2

module fig1 (a, b, c, d, e, f, g, h, O1, O2);

input a, b, c, d, e, f, g, h;

output O1, O2;

wire G1, G2, G3, G4, G5, G6, G7, G8, G9;

and G1_gate (G1, a, b);
and G2_gate (G2, c, d);
and G3_gate (G3, e, f);
nand G4_gate (G4, e, f);
or G5_gate  (G5, g, h);
and G6_gate (G6, G1, G2);
or G7_gate  (G7, G4, G5);
nand G8_gate (G8, G3, G7);
or G9_gate  (G9, G6, G7);
and G10_gate (O1, G8, G9);
or G11_gate (O2, G8, G5);

endmodule
